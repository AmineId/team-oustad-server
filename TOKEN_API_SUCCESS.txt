2. Créez .github/workflows/deploy-apk.yml :

name: Deploy Android APK

on:
  push:
    tags:
      - 'v*'

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3
      
      - name: Setup Flutter
        uses: subosito/flutter-action@v2
        with:
          flutter-version: '3.x'
      
      - name: Build APK
        run: flutter build apk --release
      
      - name: Extract version
        id: version
        run: |
          VERSION=${GITHUB_REF#refs/tags/v}
          echo "version=$VERSION" >> $GITHUB_OUTPUT
          echo "version_code=$GITHUB_RUN_NUMBER" >> $GITHUB_OUTPUT
      
      - name: Upload to Production
        run: |
          curl -X POST "https://app.team-oustad.com/android/upload" \
            -H "Authorization: Bearer ${{ secrets.OTA_API_TOKEN }}" \
            -F "apk=@build/app/outputs/flutter-apk/app-release.apk" \
            -F "version_code=${{ steps.version.outputs.version_code }}" \
            -F "version_name=${{ steps.version.outputs.version }}" \
            -F "changelog=Release ${{ steps.version.outputs.version }}" \
            -F "channel=production"

stages:
  - build
  - deploy

build:
  stage: build
  image: cirrusci/flutter:stable
  script:
    - flutter build apk --release
  artifacts:
    paths:
      - build/app/outputs/flutter-apk/app-release.apk

deploy:
  stage: deploy
  image: curlimages/curl:latest
  only:
    - tags
  script:
    - |
      curl -X POST "https://app.team-oustad.com/android/upload" \
        -H "Authorization: Bearer $OTA_API_TOKEN" \

        -F "version_name=$CI_COMMIT_TAG" \
        -F "changelog=Release $CI_COMMIT_TAG" \
        -F "channel=production"
